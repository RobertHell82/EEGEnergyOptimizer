def _build_markdown(self, snap: Snapshot, decision: Decision) -> str:
    lines = []
    lines.append(f"**Status:** {decision.zustand}")

        lines.append("### Diagnose (Gründe)")
        for key in decision.reasons:
            lines.append(f"- {REASON_LABELS_DE.get(key, key)}")
    ...
