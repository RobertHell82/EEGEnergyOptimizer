    TelemetryBuffer (telemetry_buffer.py):
      Two responsibilities, two Store keys:
        1. Identity Store (`STORAGE_TELEMETRY`):
             {"installation_id": "<uuid>", "api_key": "<base64url>", "registered_at": "<iso>"}
           - Empty file ⇒ identity_known() returns False.
        2. Event-buffer Store (`STORAGE_TELEMETRY_BUFFER`):
             [{"endpoint": "/v1/state-change", "body": {...}, "queued_at": "<iso>"}, ...]
           - Capped at TELEMETRY_BUFFER_MAX entries (FIFO, drop-oldest on overflow).

      API:
        async load() → None  (called once at construction; populates internal state from disk)
        identity_known() → bool
        get_identity() → dict | None  ({"installation_id", "api_key", "registered_at"})
        async set_identity(installation_id, api_key, registered_at) → None  (saves)
        async clear_identity() → None  (deletes the identity file via async_remove if available; falls back to async_save({}) — empty dict)
        async append(endpoint: str, body: dict) → None
            – Adds {"endpoint", "body", "queued_at": now_iso()} at tail.
            – If len > TELEMETRY_BUFFER_MAX, drop oldest (deque maxlen semantics).
            – Saves the buffer to disk on every append (Persistenz, D-06).
        peek_batch(limit: int) → list[dict]
            – Returns up to `limit` oldest entries (no removal).
        async drop(n: int) → None
            – Removes the first `n` entries (after a successful flush).
        async clear_buffer() → None
            – Empties the buffer + saves.
        size() → int

      Persistence: identity must survive HA restart — after `set_identity` then a fresh
      `TelemetryBuffer.load()`, `identity_known()` returns True with the same values.

    ```python
    """Persistent storage for telemetry identity + pending event buffer (D-04, D-05, D-06).

    Two Store files are used so a corrupted buffer cannot wipe the identity:
      - STORAGE_TELEMETRY        : identity_id, api_key, registered_at
      - STORAGE_TELEMETRY_BUFFER : FIFO ring buffer of pending events (max 100)
    """
    from __future__ import annotations

    import collections
    import logging
    from datetime import datetime, timezone
    from typing import Any

    from .const import (
        STORAGE_TELEMETRY,
        STORAGE_TELEMETRY_BUFFER,
        TELEMETRY_BUFFER_MAX,
    )

    _LOGGER = logging.getLogger(__name__)

    # HA imports guarded for test environment
    try:
        from homeassistant.helpers.storage import Store
    except ImportError:  # pragma: no cover
        Store = None  # type: ignore[assignment,misc]


    def _now_iso() -> str:
        return datetime.now(tz=timezone.utc).isoformat()


    class TelemetryBuffer:
        """Owns the persistent identity + FIFO event buffer for telemetry.

        All methods are async because Store I/O is async. Internal state is
        eagerly loaded once via load() and saved on every mutation.
        """

        def __init__(self, hass: Any) -> None:
            self._hass = hass
            self._identity_store: Any = (
                Store(hass, 1, STORAGE_TELEMETRY) if Store is not None else None
            )
            self._buffer_store: Any = (
                Store(hass, 1, STORAGE_TELEMETRY_BUFFER) if Store is not None else None
            )
            self._identity: dict | None = None
            self._buffer: collections.deque = collections.deque(maxlen=TELEMETRY_BUFFER_MAX)

        async def load(self) -> None:
            # Read identity
            if self._identity_store is not None:
                try:
                    raw = await self._identity_store.async_load()
                    if isinstance(raw, dict) and raw.get("installation_id"):
                        self._identity = raw
                except Exception:
                    _LOGGER.exception("Telemetry identity load failed")
            # Read buffer
            if self._buffer_store is not None:
                try:
                    raw = await self._buffer_store.async_load()
                    if isinstance(raw, list):
                        for entry in raw[-TELEMETRY_BUFFER_MAX:]:
                            self._buffer.append(entry)
                except Exception:
                    _LOGGER.exception("Telemetry buffer load failed")

        # ----- identity -------------------------------------------------
        def identity_known(self) -> bool:
            return self._identity is not None and bool(self._identity.get("installation_id"))

        def get_identity(self) -> dict | None:
            return dict(self._identity) if self._identity else None

        async def set_identity(self, installation_id: str, api_key: str,
                               registered_at: str) -> None:
            self._identity = {
                "installation_id": installation_id,
                "api_key": api_key,
                "registered_at": registered_at,
            }
            if self._identity_store is not None:
                await self._identity_store.async_save(self._identity)

        async def clear_identity(self) -> None:
            self._identity = None
            if self._identity_store is not None:
                # Prefer async_remove if available (HA 2024.x); else save empty dict.
                remove = getattr(self._identity_store, "async_remove", None)
                if remove is not None:
                    try:
                        await remove()
                        return
                    except Exception:
                        pass
                await self._identity_store.async_save({})

        # ----- event buffer --------------------------------------------
        async def append(self, endpoint: str, body: dict) -> None:
            self._buffer.append({
                "endpoint": endpoint,
                "body": body,
                "queued_at": _now_iso(),
            })
            await self._save_buffer()

        def peek_batch(self, limit: int) -> list[dict]:
            return list(self._buffer)[:max(limit, 0)]

        async def drop(self, n: int) -> None:
            for _ in range(min(n, len(self._buffer))):
                self._buffer.popleft()
            await self._save_buffer()

        async def clear_buffer(self) -> None:
            self._buffer.clear()
            await self._save_buffer()

        def size(self) -> int:
            return len(self._buffer)

        async def _save_buffer(self) -> None:
            if self._buffer_store is None:
                return
            try:
                await self._buffer_store.async_save(list(self._buffer))
            except Exception:
                _LOGGER.exception("Telemetry buffer save failed")
    ```

    TelemetryReporter (telemetry.py):

      Construction:
        TelemetryReporter(hass, buffer: TelemetryBuffer)
        - Reads TELEMETRY_BACKEND_URL and TELEMETRY_BOOTSTRAP_TOKEN at __init__ from const.
        - is_configured: bool — True iff both URL and token are non-empty (D-01, D-02).
          When False, all public methods are no-ops (silently return None / no exception).
        - Uses `homeassistant.helpers.aiohttp_client.async_get_clientsession(hass)` lazily
          (first request) — never closes the session (HA owns it).

      Public methods (one per backend endpoint, plus lifecycle):

        async register(profile: dict) -> bool
          - POST {url}/v1/register with header `X-Bootstrap-Token: <token>` and body filtered
            via the profile-shaping helper (settings whitelist applied).
          - On 201, parse {installation_id, api_key}, await buffer.set_identity(...).
          - Returns True iff registration succeeded. Failure does NOT buffer (no point — without
            identity, no later request can be sent anyway).

        async forget() -> bool
          - DELETE {url}/v1/installation with `Authorization: Bearer <api_key>`.
          - Always clears identity + buffer locally afterwards (D-31), even if the DELETE
            call fails (logged as a warning).
          - Returns True iff the DELETE call returned 2xx.

        async send_state_change(payload: StateChangePayload-shaped dict) -> None
          - POST {url}/v1/state-change with Bearer auth.
          - On failure: append to buffer, schedule backoff.
          - On success: drain up to TELEMETRY_FLUSH_BATCH buffered events FIFO.

        async send_snapshot(payload: SnapshotPayload-shaped dict) -> None
          - Single-snapshot variant. POST as a single object (endpoint accepts both shapes).

        async send_snapshot_batch(payloads: list[SnapshotPayload]) -> None
          - List variant. Splits into chunks of ≤100 (endpoint validation cap, endpoints.ts
            line 121). POST each chunk as an array.

        async send_outcome(payload: OutcomePayload-shaped dict) -> None
        async send_failure(payload: FailurePayload-shaped dict) -> None
        async update_profile(profile: dict) -> None  # POST /v1/profile

        async flush_buffer() -> int
          - Idempotent drain: take peek_batch(TELEMETRY_FLUSH_BATCH), POST each in order,
            await buffer.drop(n) where n is the count of consecutive successes from the head.
            Returns count drained.

      Profile shaping (used by register + update_profile):
        _shape_profile(profile: dict) -> dict
          - Allowed top-level keys are exactly the field names in ProfilePayload (types.ts):
              integration_started_at, app_version, ha_version, inverter_type,
              battery_capacity_kwh, pv_peak_kwp, forecast_provider, country_iso, settings
          - For `settings`, only TELEMETRY_SETTINGS_KEYS are kept; everything else is dropped.
          - Any unrecognized top-level key is dropped (no errors — silent filter).
          - Empty `settings` ⇒ key set to None (matches Optional<...>|null in types.ts).

      Retry / backoff (D-34, D-35, D-36):
        - Per-request: aiohttp.ClientTimeout(total=TELEMETRY_HTTP_TIMEOUT). On
          aiohttp.ClientError, asyncio.TimeoutError, or HTTP status ≥ 500, do ONE immediate
          retry (D-34). If the retry also fails, append to buffer.
        - 4xx (auth_invalid, bad_request, etc.): do NOT retry, do NOT buffer — log warning.
          Rationale: 401 = revoked api_key (forget already happened on the backend side?),
          400 = malformed body (our bug, retrying won't help). Buffer would just consume space.
        - On 429 (rate-limited): respect Retry-After header if present; otherwise backoff.
          Append to buffer.
        - Backoff state: `_consecutive_failures: int`, `_next_attempt_at: datetime|None`.
          On consecutive 5xx/network failures, double the backoff
          (TELEMETRY_BACKOFF_MIN_S × 2^(failures-1), capped at TELEMETRY_BACKOFF_MAX_S).
          On first success, reset to 0.
        - Public sends are no-ops while `now < _next_attempt_at` — they go straight to
          buffer.append without an HTTP attempt.

      Bearer-auth precondition:
        - All send_*/forget/update_profile require buffer.identity_known() == True.
          Otherwise: log debug + return without action (do NOT buffer — no api_key means no
          recovery path; the panel must call register first).

    ```python
    """TelemetryReporter — HTTP client + retry/backoff + whitelist filter (D-01..D-36).

    All public coroutines are safe to call repeatedly; if the reporter is not configured
    (URL or bootstrap token empty), every method is a silent no-op (D-01, D-02).
    """
    from __future__ import annotations

    import asyncio
    import logging
    from datetime import datetime, timedelta, timezone
    from typing import Any

    from .const import (
        TELEMETRY_BACKEND_URL,
        TELEMETRY_BOOTSTRAP_TOKEN,
        TELEMETRY_BACKOFF_MAX_S,
        TELEMETRY_BACKOFF_MIN_S,
        TELEMETRY_BUFFER_MAX,
        TELEMETRY_FLUSH_BATCH,
        TELEMETRY_HTTP_TIMEOUT,
        TELEMETRY_SETTINGS_KEYS,
    )
    from .telemetry_buffer import TelemetryBuffer

    _LOGGER = logging.getLogger(__name__)

    # HA imports guarded for test environment
    try:
        import aiohttp
        from homeassistant.helpers.aiohttp_client import async_get_clientsession
    except ImportError:  # pragma: no cover
        aiohttp = None  # type: ignore
        async_get_clientsession = None  # type: ignore[assignment]


    _PROFILE_TOP_LEVEL = (
        "integration_started_at", "app_version", "ha_version", "inverter_type",
        "battery_capacity_kwh", "pv_peak_kwp", "forecast_provider", "country_iso",
        "settings",
    )


    def _now_utc() -> datetime:
        return datetime.now(tz=timezone.utc)


    class TelemetryReporter:
        def __init__(self, hass: Any, buffer: TelemetryBuffer) -> None:
            self._hass = hass
            self._buffer = buffer
            self._url = (TELEMETRY_BACKEND_URL or "").rstrip("/")
            self._bootstrap = TELEMETRY_BOOTSTRAP_TOKEN or ""
            self._consecutive_failures = 0
            self._next_attempt_at: datetime | None = None
            self._send_lock = asyncio.Lock()
            self._session: Any = None  # populated lazily

        @property
        def is_configured(self) -> bool:
            return bool(self._url and self._bootstrap)

        # ----- profile filter ------------------------------------------
        @staticmethod
        def _shape_profile(profile: dict) -> dict:
            out: dict = {}
            for key in _PROFILE_TOP_LEVEL:
                if key in profile:
                    out[key] = profile[key]
            settings = profile.get("settings")
            if isinstance(settings, dict):
                out["settings"] = {
                    k: v for k, v in settings.items() if k in TELEMETRY_SETTINGS_KEYS
                } or None
            else:
                out["settings"] = None
            return out

        # ----- public sends --------------------------------------------
        async def register(self, profile: dict) -> bool:
            if not self.is_configured:
                return False
            shaped = self._shape_profile(profile)
            res = await self._post(
                "/v1/register", shaped,
                headers={"X-Bootstrap-Token": self._bootstrap},
                expect_status=201, buffer_on_failure=False, retry_on_5xx=True,
            )
            if not res or not res.get("ok"):
                return False
            data = res.get("body") or {}
            inst = data.get("installation_id")
            api_key = data.get("api_key")
            if not inst or not api_key:
                return False
            await self._buffer.set_identity(inst, api_key, _now_utc().isoformat())
            return True

        async def forget(self) -> bool:
            if not self.is_configured or not self._buffer.identity_known():
                # Local cleanup even if not configured / no identity (idempotent)
                await self._buffer.clear_identity()
                await self._buffer.clear_buffer()
                return False
            ok = await self._delete_authed("/v1/installation")
            await self._buffer.clear_identity()
            await self._buffer.clear_buffer()
            return ok

        async def send_state_change(self, payload: dict) -> None:
            await self._send_authed("/v1/state-change", payload)

        async def send_snapshot(self, payload: dict) -> None:
            await self._send_authed("/v1/snapshot", payload)

        async def send_snapshot_batch(self, payloads: list[dict]) -> None:
            if not payloads:
                return
            # Backend caps at 100 per request (endpoints.ts line 121)
            for i in range(0, len(payloads), 100):
                chunk = payloads[i:i + 100]
                await self._send_authed("/v1/snapshot", chunk)

        async def send_outcome(self, payload: dict) -> None:
            await self._send_authed("/v1/outcome", payload)

        async def send_failure(self, payload: dict) -> None:
            await self._send_authed("/v1/failure", payload)

        async def update_profile(self, profile: dict) -> None:
            await self._send_authed("/v1/profile", self._shape_profile(profile))

        async def flush_buffer(self) -> int:
            if not self.is_configured or not self._buffer.identity_known():
                return 0
            batch = self._buffer.peek_batch(TELEMETRY_FLUSH_BATCH)
            drained = 0
            for entry in batch:
                ok = await self._post_authed(entry["endpoint"], entry["body"],
                                              already_buffered=True)
                if not ok:
                    break
                drained += 1
            if drained > 0:
                await self._buffer.drop(drained)
            return drained

        # ----- internals -----------------------------------------------
        async def _send_authed(self, endpoint: str, body) -> None:
            if not self.is_configured:
                return
            if not self._buffer.identity_known():
                _LOGGER.debug("Telemetry: skip %s — no identity", endpoint)
                return
            # Backoff gate: while in backoff, append to buffer instead of attempting
            if self._next_attempt_at is not None and _now_utc() < self._next_attempt_at:
                if isinstance(body, dict):
                    await self._buffer.append(endpoint, body)
                return
            ok = await self._post_authed(endpoint, body, already_buffered=False)
            if ok:
                # Drain up to TELEMETRY_FLUSH_BATCH from buffer FIFO
                await self.flush_buffer()

        async def _post_authed(self, endpoint: str, body, *, already_buffered: bool) -> bool:
            ident = self._buffer.get_identity()
            if not ident:
                return False
            headers = {"Authorization": f"Bearer {ident['api_key']}"}
            res = await self._post(endpoint, body, headers=headers,
                                    expect_status={200, 204},
                                    buffer_on_failure=not already_buffered,
                                    retry_on_5xx=True)
            return bool(res and res.get("ok"))

        async def _delete_authed(self, endpoint: str) -> bool:
            ident = self._buffer.get_identity()
            if not ident:
                return False
            headers = {"Authorization": f"Bearer {ident['api_key']}"}
            session = self._get_session()
            url = f"{self._url}{endpoint}"
            try:
                async with session.delete(
                    url, headers=headers,
                    timeout=aiohttp.ClientTimeout(total=TELEMETRY_HTTP_TIMEOUT),
                ) as resp:
                    return 200 <= resp.status < 300
            except (asyncio.TimeoutError, aiohttp.ClientError):
                _LOGGER.warning("Telemetry forget: network error")
                return False

        async def _post(self, endpoint, body, *, headers, expect_status,
                        buffer_on_failure, retry_on_5xx) -> dict | None:
            """Single POST with one-time retry on 5xx/network and backoff bookkeeping.

            Returns:
              {"ok": True, "body": <dict|None>} on success
              {"ok": False, "status": <int|None>} on failure
              None if the reporter is unconfigured (caller already guards this).
            """
            if not self.is_configured or aiohttp is None:
                return None
            url = f"{self._url}{endpoint}"
            session = self._get_session()
            timeout = aiohttp.ClientTimeout(total=TELEMETRY_HTTP_TIMEOUT)

            attempts = 2 if retry_on_5xx else 1
            last_status: int | None = None
            for attempt in range(attempts):
                try:
                    async with session.post(url, json=body, headers=headers,
                                             timeout=timeout) as resp:
                        last_status = resp.status
                        # 4xx → no retry, no buffer
                        if 400 <= resp.status < 500 and resp.status != 429:
                            self._on_4xx(endpoint, resp.status)
                            return {"ok": False, "status": resp.status}
                        if (isinstance(expect_status, set) and resp.status in expect_status) \
                                or (isinstance(expect_status, int) and resp.status == expect_status):
                            self._on_success()
                            try:
                                if resp.content_length and resp.content_length > 0:
                                    return {"ok": True, "body": await resp.json()}
                            except Exception:
                                pass
                            return {"ok": True, "body": None}
                        # 429 → respect Retry-After + buffer
                        if resp.status == 429:
                            ra = resp.headers.get("Retry-After")
                            self._on_5xx_or_rate_limited(retry_after_s=int(ra) if ra and ra.isdigit() else None)
                            break
                        # 5xx → retry once
                        if resp.status >= 500 and attempt + 1 < attempts:
                            continue
                        self._on_5xx_or_rate_limited()
                        break
                except (asyncio.TimeoutError, aiohttp.ClientError) as exc:
                    last_status = None
                    if attempt + 1 < attempts:
                        continue
                    _LOGGER.debug("Telemetry %s network error: %s", endpoint, exc)
                    self._on_5xx_or_rate_limited()
                    break

            if buffer_on_failure and isinstance(body, dict):
                await self._buffer.append(endpoint, body)
            return {"ok": False, "status": last_status}

        def _on_success(self) -> None:
            self._consecutive_failures = 0
            self._next_attempt_at = None

        def _on_4xx(self, endpoint: str, status: int) -> None:
            _LOGGER.warning("Telemetry %s rejected with %d (no retry, no buffer)",
                            endpoint, status)
            # 4xx does NOT advance backoff — it's a permanent error.

        def _on_5xx_or_rate_limited(self, *, retry_after_s: int | None = None) -> None:
            self._consecutive_failures += 1
            if retry_after_s is not None:
                delay = max(1, min(retry_after_s, TELEMETRY_BACKOFF_MAX_S))
            else:
                exp = TELEMETRY_BACKOFF_MIN_S * (2 ** (self._consecutive_failures - 1))
                delay = min(exp, TELEMETRY_BACKOFF_MAX_S)
            self._next_attempt_at = _now_utc() + timedelta(seconds=delay)

        def _get_session(self):
            if self._session is None and async_get_clientsession is not None:
                self._session = async_get_clientsession(self._hass)
            return self._session
    ```
