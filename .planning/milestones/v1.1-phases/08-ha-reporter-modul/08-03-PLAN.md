      def _normalize_state(zustand: str | None) -> str | None:
          """Convert a Decision.zustand label into lowercase snake_case.

          The integration uses German labels with hyphens and umlauts internally
          ('Morgen-Einspeisung', 'Abend-Entladung', 'Normal'). The backend (and
          Phase 9 dashboard JOINs) require a single canonical form. This helper
          is the ONE place that performs the conversion. Used by:
            - state-change `transition` strings (W-2)
            - snapshot.state field (W-6)
            - outcome.event_type field (W-2)
            - block_predictions dict key (W-2)
          """
          if zustand is None:
              return None
          return (zustand.lower()
                          .replace(" ", "_")
                          .replace("-", "_"))

      Examples:
        "Normal"             → "normal"
        "Morgen-Einspeisung" → "morgen_einspeisung"
        "Abend-Entladung"    → "abend_entladung"

        def _trapezoid_kwh(samples: list[tuple[datetime, float]]) -> float:
            """Trapezoidal integration of power (kW) over time → energy (kWh).

            Formula: sum(((p[i] + p[i+1]) / 2) * dt_hours)
            where dt_hours = (samples[i+1].ts - samples[i].ts).total_seconds() / 3600.

            Returns 0.0 for empty / single-sample input. Filters None power values
            before integrating. Sorts by ts so out-of-order samples integrate cleanly.
            """
            valid = sorted(
                [(ts, p) for ts, p in samples if p is not None],
                key=lambda x: x[0],
            )
            if len(valid) < 2:
                return 0.0
            total = 0.0
            for i in range(len(valid) - 1):
                dt_hours = (valid[i + 1][0] - valid[i][0]).total_seconds() / 3600.0
                if dt_hours <= 0:
                    continue
                total += ((valid[i][1] + valid[i + 1][1]) / 2.0) * dt_hours
            return total

          #   stats_key="evening" → event_type="abend_entladung"
          from .const import STATE_MORGEN_EINSPEISUNG, STATE_ABEND_ENTLADUNG
          from . import _normalize_state   # one canonicalization function (W-2)
          if stats_key == "morning":
              event_type = _normalize_state(STATE_MORGEN_EINSPEISUNG)  # "morgen_einspeisung"
          elif stats_key == "evening":
              event_type = _normalize_state(STATE_ABEND_ENTLADUNG)     # "abend_entladung"
          else:
              return  # unknown session — don't emit

          predictions = (self._data or {}).get("block_predictions", {}).get(event_type)
          # Pull SOC end from current battery_soc state via _read_float — but
          # accept that this is approximate (the optimizer's snapshot SOC at
          # block-end is more accurate; if a recent decision is available via
          # data["optimizer"].last_decision, prefer that).
          last_dec = (self._data or {}).get("optimizer")
          soc_end = None
          if last_dec is not None and getattr(last_dec, "last_decision", None) is not None:
              soc_end = last_dec.last_decision.snapshot.get("soc_pct")

          # ---- W-1: compute peak_power_kw + actual_pv_kwh + actual_consumption_kwh ----
          # Read the SHARED snapshot_queue populated by the 30-min snapshot tick.
          # Filter to samples within [started_at, ended_at] of THIS block.
          peak_power_kw = None
          actual_pv_kwh = None
          actual_consumption_kwh = None
          ended_at_dt = now_local.astimezone(timezone.utc)
          started_at_str = (predictions or {}).get("started_at")
          if started_at_str:
              try:
                  started_at_dt = datetime.fromisoformat(started_at_str.replace("Z", "+00:00"))
              except Exception:
                  started_at_dt = None
              if started_at_dt is not None:
                  queue = (self._data or {}).get("snapshot_queue") or []
                  window = []
                  for snap in queue:
                      ts_raw = snap.get("ts")
                      if not ts_raw:
                          continue
                      try:
                          ts_dt = datetime.fromisoformat(str(ts_raw).replace("Z", "+00:00"))
                      except Exception:
                          continue
                      if started_at_dt <= ts_dt <= ended_at_dt:
                          window.append((ts_dt, snap))
                  # peak_power_kw — fall back to None if no usable snapshots
                  grids = [abs(s["grid_now_kw"]) for _, s in window
                           if s.get("grid_now_kw") is not None]
                  if grids:
                      peak_power_kw = round(max(grids), 3)
                  # Trapezoid actuals — fall back to None if <2 samples
                  pv_samples = [(ts, s.get("pv_now_kw")) for ts, s in window
                                if s.get("pv_now_kw") is not None]
                  cons_samples = [(ts, s.get("consumption_now_kw")) for ts, s in window
                                  if s.get("consumption_now_kw") is not None]
                  if len(pv_samples) >= 2:
                      actual_pv_kwh = round(_trapezoid_kwh(pv_samples), 3)
                  if len(cons_samples) >= 2:
                      actual_consumption_kwh = round(_trapezoid_kwh(cons_samples), 3)

          payload = {
              "event_type": event_type,
              "started_at": started_at_str if started_at_str else session.get("start_utc", ""),
              "ended_at": ended_at_dt.isoformat(),
              "duration_minutes": int(duration_min),
              "grid_export_kwh": float(round(kwh, 3)),
              "peak_power_kw": peak_power_kw,                                 # W-1: computed
              "soc_start_pct": predictions.get("soc_start_pct") if predictions else None,
              "soc_end_pct": int(soc_end) if soc_end is not None else None,
              "predicted_pv_kwh": predictions.get("predicted_pv_kwh") if predictions else None,
              "actual_pv_kwh": actual_pv_kwh,                                 # W-1: computed
              "predicted_consumption_kwh": predictions.get("predicted_consumption_kwh") if predictions else None,
              "actual_consumption_kwh": actual_consumption_kwh,               # W-1: computed
              "terminated_by": "block_end",
          }
          # Pop predictions so a follow-on session of the same kind doesn't reuse stale predictions
          if isinstance(self._data, dict):
              self._data.get("block_predictions", {}).pop(event_type, None)
          # Fire-and-forget — Reporter handles buffer/retry on failure
          self._hass.async_create_task(self._reporter.send_outcome(payload))
